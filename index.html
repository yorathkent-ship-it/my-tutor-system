部署指南 & Google 試算表專屬 App Script

  1. 開啟您的 Google 試算表。
  2. 選單中點選 「擴充功能」 -> 「Apps Script」。
  3. 覆蓋原本所有程式碼，貼上下方程式碼。
  4. 點擊 「部署」 -> 「新增部署」。
  5. 類型選 「網路應用程式」：負責人選自己、誰能存取選 「所有人 (Anyone)」。
// 專屬四堂一期、計費與點名同步 Apps Script
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var students = [];
  
  for(var i = 1; i < data.length; i++) {
    students.push({
      id: data[i][0].toString(),
      name: data[i][1],
      phone: data[i][2].toString(),
      balance: Number(data[i][3]),
      fee: Number(data[i][4]) || 150,
      attendance: data[i][5] ? JSON.parse(data[i][5]) : {}
    });
  }
  return ContentService.createTextOutput(JSON.stringify(students))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var params = JSON.parse(e.postData.contents);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  
  if (params.action === "syncAll") {
    sheet.clear();
    sheet.appendRow(["學號", "姓名", "聯絡電話", "課堂餘額(Balance)", "單堂或每期學費", "詳細出席時間記錄"]);
    params.students.forEach(function(st) {
      sheet.appendRow([st.id, st.name, st.phone, st.balance, st.fee, JSON.stringify(st.attendance)]);
    });
    return ContentService.createTextOutput(JSON.stringify({status: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({status: "error"}))
    .setMimeType(ContentService.MimeType.JSON);
}
